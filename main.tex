\begin{center}
{\Large \textbf{摘\quad 要}}
\end{center}
\vspace{0.2cm}

面对人口老龄化加速背景下社区养老服务"建在哪、建多大、收多少钱"的全链条决策难题，本文以某城市10个连片小区为实证场景，构建"预测--选址--定价--灵敏度"四阶段递进求解框架，在120万元总预算与1000m服务半径硬约束下完成从需求预测到方案落地的闭环求解。

\textbf{问题1——人口与服务需求预测}：构造增生型3状态Markov转移矩阵，以7\%年注入率与5\%死亡率刻画系统开放性；同步建立Leslie矩阵进行双模型三角校验，两范式对第5年末60+人口的预测偏差仅1.7\%，Leslie主特征值$\lambda_1=1.0198$与年均增长率2.0\%精确吻合。核心发现：失能老人因棘轮效应从625人飙升至1,125人（+80.3\%），失能率从9.1\%攀升至14.8\%；半失能群体充当自理向失能的加速通道。消费约束预诊断表明失能老人全10小区触发等比例削减，缺口23.9\%--49.1\%。

\textbf{问题2——选址与规模优化}：将Church\&ReVelle经典MCLP扩展为MILP-MCLP-ES（嵌入式最大覆盖选址）模型，针对满意度函数中$S_2$与$x_{ij}$的非线性乘积引入McCormick包络，$S_1$（4段）与$S_2$（5段）阶梯函数以Big-M方法精确线性化，CBC分支定界在120秒内收敛至gap$<1\%$的确定解。最优方案F(大型,45万)+H(中型,32万)+J(中型,32万)，总成本109万元，10/10全覆盖，加权满意度0.850，年总利润1,097.7万元。三站平均利用率94.3\%，确定解目标值54.225较非确定解提升24\%。

\textbf{问题3——服务定价与政府补贴}：在固化Q2选址方案的基础上，构建词典序MILP定价模型——优先最大化加权$S_3$，次优先以弱激励$\varepsilon=10^{-4}$最大化营收，受利润率$\le8\%$监管红线与补贴上限双重约束。全6项服务$S_3=1.000$平价，仅上门护理从30元降至12.64元（降幅57.9\%）以吸收F站8\%利润率紧约束，体现Ramsey-Boiteux反向价格歧视原理。三层交叉补贴使紧急救助年净支出47.42万元由营利利润331.3万元完全吸收，交叉补贴率仅14.3\%。

\textbf{问题4——灵敏度与鲁棒性}：对预算总额、建设成本、转移概率三类外部冲击实施三场景6组MILP重求解。场景A预算释放下满意度单调递增但边际递减，130万为甜点门槛；场景B成本通胀20\%击穿全覆盖防线，覆盖率降至80\%；场景C银发海啸下系统从3站自适应跃迁至4站，维持90\%覆盖率。多维鲁棒性韧性矩阵显示各维度$\rho>0.80$，系统鲁棒性为强。该框架可迁移至分级诊疗、冷链前置仓、充电桩等公共设施优化场景。

\textbf{关键词：}嵌入式养老；Markov链；最大覆盖选址（MCLP）；混合整数线性规划（MILP）；McCormick包络；Ramsey-Boiteux定价；交叉补贴；鲁棒性分析

\section{结论}

本文面向嵌入式社区养老服务站"建在哪、建多大、收多少钱、抗多大风险"的全链条工程决策问题，以增生型Markov链、MILP-MCLP-ES选址模型与词典序MILP定价模型为核心工具，在10个小区实证场景上完成了预测--选址--定价--灵敏度的闭环求解，核心结论如下：

\begin{enumerate}[label=\textbf{(\arabic*)}, leftmargin=*]
    \item \textbf{双模型三角校验确立基线可信度}。Markov链与Leslie矩阵对第5年末60+人口的预测偏差仅1.7\%，$\lambda_1=1.0198$。失能老人因棘轮效应5年激增+80.3\%（625$\to$1,125人），失能率从9.1\%升至14.8\%。
    \item \textbf{MILP-MCLP-ES实现全覆盖}。F(大型,45万)+H(中型,32万)+J(中型,32万)，成本109万，10/10全覆盖，目标值54.225较非确定解提升24\%。三站利用率92--96\%。
    \item \textbf{词典序MILP定价全$S_3=1.000$}。上门护理30$\to$12.64元（-57.9\%）吸收F站8\%利润率紧约束，体现Ramsey-Boiteux反向价格歧视。紧急救助年亏47.42万元由营利利润331.3万元完全吸收，交叉补贴率14.3\%。
    \item \textbf{多维扰动下韧性系数均$>$0.80}。三场景6组MILP重求解验证强鲁棒性，预算120$\to$130万触发覆盖率相变80\%$\to$100\%，ROI=22.4:1。
\end{enumerate}

MILP-MCLP-ES框架——Church\&ReVelle MCLP底座、McCormick/Big-M线性化、多起点重启、三层交叉补贴——可平滑迁移至分级诊疗、冷链前置仓、充电桩等公共设施优化场景。

\section{主要算法伪代码}

\subsection*{算法1：MILP-MCLP-ES选址-规模优化（对应§3.2）}

\begin{algorithm}[H]
\caption{MILP-MCLP-ES核心算法框架}
\label{alg:mclp_es}
\begin{algorithmic}[1]
\Require 小区集 $\mathcal{I}$，规模集 $\mathcal{K}$，需求矩阵 $\mathbf{D}$，距离矩阵 $d_{ij}$，预算 $B=120$万
\Ensure 最优选址-规模方案 $\mathbf{y}^*=\{y_{ik}^*\}$，最优服务分配 $\mathbf{x}^*=\{x_{ij}^*\}$
\State \textbf{// Step 1: 构建决策变量}
\State $y_{ik} \in \{0,1\}$ \Comment{小区$i$是否建规模$k$的站}
\State $x_{ij} \in \{0,1\}$ \Comment{$j$小区是否由$i$站服务}
\State $\delta_{ij\ell}^{S1}\in\{0,1\}$ \Comment{$S_1$距离满意度4段指示变量}
\State $\delta_{i\ell}^{S2}\in\{0,1\}$ \Comment{$S_2$响应满意度5段指示变量}
\State $w_{ij}\in[0,1]$ \Comment{McCormick辅助变量，替代$S2_i\cdot x_{ij}$}
\State \textbf{// Step 2: 目标函数（双目标加权聚合）}
\State $\max \quad \lambda\cdot\text{Coverage} + (1-\lambda)\cdot\text{AvgSatisfaction}$
\State \textbf{// Step 3: Big-M线性化$S_1$（4段阶梯，断点0/300/500/650/1000m）}
\For{每对$(i,j)$满足$d_{ij}\le 1000$m}
    \State $\sum_{\ell=1}^{4}\delta_{ij\ell}^{S1}=x_{ij}$
    \State $S1_{ij}=1.00\delta_1+0.90\delta_2+0.75\delta_3+0.60\delta_4$
\EndFor
\State \textbf{// Step 4: McCormick包络线性化$S_2\cdot x_{ij}$}
\For{每站$i$，每小区$j$}
    \State $\sum_{\ell=1}^{5}\delta_{i\ell}^{S2}=\sum_k y_{ik}$； $S2_i=1.00\delta_1+0.93\delta_2+0.85\delta_3+0.72\delta_4+0.60\delta_5$
    \State $w_{ij}\le x_{ij}$； $w_{ij}\le S2_i$； $w_{ij}\ge S2_i-(1-x_{ij})$； $0\le w_{ij}\le 1$
\EndFor
\State \textbf{// Step 5: 约束条件}
\State $\sum_i\sum_k C_k^{\text{build}}y_{ik}\le B$ \Comment{预算约束}
\State $\sum_i x_{ij}=1,\;\forall j$ \Comment{每位老人由最近站服务}
\State $\sum_j\sum_e D_{je}\cdot x_{ij}\le 30\cdot\sum_k Q_k^{\max}y_{ik},\;\forall i$ \Comment{容量约束}
\State $x_{ij}=0,\;\forall i,j: d_{ij}>1000$ \Comment{服务半径约束}
\State \textbf{// Step 6: 分支定界求解}
\State 调用CBC求解器（time\_limit=120s, randomSeed=42, gapRel=0.01）
\State \Return $\mathbf{y}^*$, $\mathbf{x}^*$, 目标值$Z^*$
\end{algorithmic}
\end{algorithm}

\subsection*{算法2：S3价格满意度Big-M线性化（对应§3.3）}

\begin{algorithm}[H]
\caption{定价优化S3满意度Big-M线性化}
\label{alg:s3_bigm}
\begin{algorithmic}[1]
\Require 服务基准价 $\mathbf{p}^{\text{base}}$；S3断点 $\{1.0, 1.1, 1.2\}\times\mathbf{p}^{\text{base}}$；S3分值 $\{1.00, 0.90, 0.75, 0.60\}$
\Ensure 最优定价 $\mathbf{p}^*=\{p_s^*\}_{s\in\mathcal{S}}$；S3满意度 $\{S3_s\}$
\State \textbf{// Step 1: 4段指示变量}
\For{每项服务$s$（紧急救助除外）}
    \State $\sum_{\ell=0}^{3}\delta_{s\ell}^{S3}=1$ \Comment{恰好选一段}
    \For{$\ell=0$ \textbf{to} $3$}
        \State $p_s\ge\text{lo}_\ell - M(1-\delta_{s\ell}^{S3})$； $p_s\le\text{hi}_\ell + M(1-\delta_{s\ell}^{S3})$
    \EndFor
    \State $S3_s=\sum_\ell \text{val}_\ell\cdot\delta_{s\ell}^{S3}$
\EndFor
\State \textbf{// Step 2: 词典序目标}
\State $\max \quad \underbrace{\sum_s D_s\cdot S3_s}_{\text{优先: }S_3\text{最大化}} + \underbrace{\varepsilon\cdot\sum_s D_s\cdot p_s}_{\text{次优: 营收激励}(\varepsilon=10^{-4})}$
\State \textbf{// Step 3: 利润率约束}
\State $\Pi_i/(\text{年化成本}_i)\le 0.08,\;\forall i$ \Comment{利润率$\le$8\%红线}
\State \Return $\mathbf{p}^*$, $\{S3_s\}$, 各站利润率$\{\pi_i\}$
\end{algorithmic}
\end{algorithm}

\subsection*{算法3：三层交叉补贴与定价优化（对应§5.3.3）}

\begin{algorithm}[H]
\caption{三层交叉补贴算法}
\label{alg:cross_subsidy}
\begin{algorithmic}[1]
\Require 固化选址方案 $\mathbf{y}^*$；各站点服务需求矩阵 $\mathbf{D}$；服务基准价 $\mathbf{p}^{\text{base}}$；利润率上界 $\pi^{\max}=8\%$
\Ensure 最优定价 $\mathbf{p}^*$；各站点利润率 $\pi_i$；交叉补贴率 $\eta$
\State \textbf{// 第一阶段：站内营利养公益}
\For{每站 $i \in \mathcal{I}^{\text{built}}$}
    \State $C^{\text{welfare}}_i \gets 12 \times D_{i,\text{emergency}} \times c_{\text{emergency}}$
    \State $R^{\text{profit}}_i \gets 12 \times \sum_{s \neq \text{emergency}} D_{i,s} \times (p^{\text{base}}_s - c_s)$
\EndFor
\State \textbf{// 第二阶段：站间补贴效率再分配}
\For{每站 $i \in \mathcal{I}^{\text{built}}$}
    \State $S^{\text{actual}}_i \gets \min(r^{\text{sub}} \times \sum_{s \neq \text{emergency}} D_{i,s},\; S^{\text{cap}}_i)$
\EndFor
\State \textbf{// 第三阶段：Ramsey-Boiteux反向价格歧视}
\State 构建MILP词典序目标并求解 → $\mathbf{p}^*$；识别绑定站点$i^*$
\If{$\pi_{i^*} = \pi^{\max}$}
    \State $\mathbf{p}^*_{s^*} \gets \max\{p \mid S_3(p) = 1.000\}$
\EndIf
\State \Return $\mathbf{p}^*$, $\{\pi_i\}$, $\eta \gets \sum_i C^{\text{welfare}}_i / \sum_i R^{\text{profit}}_i$
\end{algorithmic}
\end{algorithm}

import numpy as np

# 转移矩阵 A (3x3): 自理/半失能/失能
alpha_sm, alpha_md, mu, beta = 0.045, 0.10, 0.05, 0.07
A = np.array([

S = S0.copy()  # S0: (10, 3) 基年状态矩阵

# 对应 §3.2: MILP-MCLP-ES 选址-规模优化
from pulp import LpProblem, LpMaximize, LpVariable, lpSum, LpBinary

prob = LpProblem("MCLP_ES", LpMaximize)

# 决策变量
y = {(i,k): LpVariable(f"y_{i}_{k}", cat=LpBinary) for i in I for k in K}
x = {(i,j): LpVariable(f"x_{i}_{j}", cat=LpBinary) for i in I for j in J}

# McCormick包络: w[i,j] = S2[i] * x[i,j]
w = {(i,j): LpVariable(f"w_{i}_{j}", 0, 1) for i in I for j in J}
for i in I:
    for j in J:
        prob += w[i,j] <= x[i,j]
        prob += w[i,j] <= s2[i]
        prob += w[i,j] >= s2[i] - (1 - x[i,j])

# 目标: 覆盖率×5.0 + 满意度×0.5
prob += lpSum([coverage * 5.0, weighted_satisfaction * 0.5])

# 求解 (CBC, 120s, randomSeed=42)
prob.solve(PULP_CBC_CMD(timeLimit=120, options=['randomSeed 42']))
\end{lstlisting}

\subsection*{代码B3：词典序MILP定价（对应§3.3，solve\_q3.py）}

# 对应 §3.3: 词典序MILP定价 — 优先S3最大化，次优营收
epsilon = 1e-4  # 弱激励系数

# Phase 1: 最大化S3
prob.setObjective(lpSum([demand[s] * s3[s] for s in services]))
prob.solve()

# Phase 2: 固定S3最优值，最大化营收
s3_opt = {s: s3[s].varValue for s in services}
prob += lpSum([demand[s] * s3[s] for s in services]) >= sum(
    demand[s] * s3_opt[s] for s in services)
prob.setObjective(lpSum([demand[s] * s3[s] * 0.5 +
    epsilon * demand[s] * price[s] for s in services]))
prob.solve()
\end{lstlisting}
